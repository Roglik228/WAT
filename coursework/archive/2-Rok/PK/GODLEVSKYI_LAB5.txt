def pierwsza_mod4_3(bits):
    lo, hi = 1 << (bits - 1), (1 << bits) - 1
    while True:
        a = random_prime(hi, False, lo)
        if a % 4 == 3:
            return a

def losuj_coprime(n):
    while True:
        x = randint(2, n - 2)
        if gcd(x, n) == 1:
            return x

def rzut_moneta_blum(b1, b2, case=1):
    
    p = pierwsza_mod4_3(b1)

    if case == 2:
        while True:
            q = random_prime((1 << b2) - 1, proof=False, lbound=1 << (b2 - 1))
            if q % 4 == 1:
                break
    else:
        q = pierwsza_mod4_3(b2)

    n = p * q
    x = losuj_coprime(n)
    x0 = pow(x, 2, n)
    x1 = pow(x0, 2, n)
    x1_send = (x1 + 1) % n if case == 3 else x1
    
    print("Krok 1:")
    print("A generuje parametry:")
    print(f"p = {p}")
    print(f"q = {q}")
    print(f"Liczba Bluma n = {n}")
    print(f"Losowe x = {x}\n")
    
    print("Krok 2:")
    print(f"A oblicza x0 = {x0} i x1 = {x1}")
    print(f"A wysyla do B: x1 = {x1_send} i n = {n}\n")

    print("Krok 3:")
    guess = choice(["parzyste", "nieparzyste"])
    print(f"B zgaduje: zgaduję {guess}\n")

    print("Krok 4:")
    print(f"A przesyła do B: x = {x}, x0 = {x0}, p = {p} i q = {q}\n")

    # weryfikacja
    print("Krok 5:")
    print("B sprawdza dane:")
    
    ok_blum = (p * q == n and is_prime(p) and is_prime(q) and p % 4 == 3 and q % 4 == 3)
    ok_eq = (pow(x, 2, n) == x0 and pow(x0, 2, n) == x1_send)
    
    print(f"B oblicza: n = {n},x0 = {x0} i x1 = {x1_send}")

    if not ok_blum:
        print("n nie jest liczbą Bluma – weryfikacja przerwana.")
        return
    if not ok_eq:
        print("Relacja x, x0, x1 nie zachodzi – weryfikacja przerwana.")
        return

    actual = "parzyste" if x0 % 2 == 0 else "nieparzyste"
    wynik  = "orzeł" if actual == guess else "reszka"
    print(f"Faktyczna parzystość x0: {actual}")
    print(f"Wynik rzutu: {wynik}")

# parametry
b1, b2 = 30, 31
print(f"Długość liczby p w bitach: {b1}")
print(f"Długość liczby q w bitach: {b2}\n")

print("============SYMULACJA 1 (Poprawna)============\n")
rzut_moneta_blum(b1, b2, case=1)
print("\n============SYMULACJA 2 (n nie liczba Bluma)============\n")
rzut_moneta_blum(b1, b2, case=2)
print("\n============SYMULACJA 3 (x1 jest podmienione)============\n")
rzut_moneta_blum(b1, b2, case=3)
